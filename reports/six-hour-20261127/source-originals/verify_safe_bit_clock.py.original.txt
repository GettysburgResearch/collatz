"""Build the three actual-gate modules in isolation and validate all theorem axiom sets."""
from tempfile import TemporaryDirectory
from pathlib import Path
import re
import os
import json
from verify_ordinary_safe_modules import HERE,ORIGINAL,LEAN,MATHLIB,FLAGS,run,messages,h


def main():
    files=(('PermanentSafeDebt',ORIGINAL,'7ccc0c4921083b94f01c43eb604538fb257d3b40a15b8e6bec62f99d8fd91552'),
           ('PermanentSafeOrdinary',ORIGINAL,'9e6b71ae0c5baca5e64d96c566ce8ba40b8210441ac2a27c066f6d8ddf74aa9f'),
           ('PermanentSafeBitClock',HERE,None))
    lean_pin=run([LEAN,'--githash'],HERE).strip()
    lib_pin=run(['git','-C',MATHLIB,'rev-parse','HEAD'],HERE).strip()
    if lean_pin!='5045d0056413266e57c625dcd7c365b10e377c52' or lib_pin!='d13f23b723b8a846827a245b89c10fc7d3f11612':
        raise ValueError('pin mismatch')
    if run(['git','-C',MATHLIB,'status','--porcelain','--untracked-files=no'],HERE).strip():
        raise ValueError('modified mathlib')
    declarations={};objects={};sources={};logs=[]
    with TemporaryDirectory(dir=HERE) as tmp:
        root=Path(tmp)
        if list(root.iterdir()):raise ValueError('workdir not initially empty')
        env=os.environ.copy()
        env['LEAN_PATH']=os.pathsep.join(map(str,[root,MATHLIB/'.lake/build/lib/lean',
                                                     *sorted((MATHLIB/'.lake/packages').glob('*/.lake/build/lib/lean'))]))
        for mod,where,expected in files:
            source=root/(mod+'.lean'); source.write_bytes((where/source.name).read_bytes())
            source_hash=h(source);sources[mod]=source_hash
            if expected and source_hash!=expected:raise ValueError((mod,source_hash))
            body=source.read_text()
            if re.search(r'\b(?:sorry|admit|axiom|native_decide|unsafe)\b',body):raise ValueError('placeholder')
            names={mod+'.'+x for x in re.findall(r'^(?:private\s+)?(?:theorem|lemma)\s+(\w+)',body,re.M)}
            printed=set(re.findall(r'^#print axioms ([.\w]+)',body,re.M))
            if names!=printed:raise ValueError('coverage')
            obj=root/(mod+'.olean')
            result=messages(run([LEAN,*FLAGS,'-R',root,'-o',obj,source],root,env),names)
            objects[mod]=h(obj)
            declarations.update(result)
            logs.extend({'stage':'source_build','name':name,'axioms':ax} for name,ax in sorted(result.items()))
        probe=root/'BitClockImportAudit.lean'
        probe.write_text('import PermanentSafeBitClock\n'+'\n'.join('#print axioms '+n for n in sorted(declarations))+'\n')
        imported=messages(run([LEAN,*FLAGS,'-R',root,probe],root,env),declarations)
        if imported!=declarations:raise ValueError('mismatched fresh olean import')
        logs.extend({'stage':'object_import','name':name,'axioms':ax} for name,ax in sorted(imported.items()))
    logfile=HERE/'SAFE_BIT_CLOCK_AXIOMS.jsonl'
    logfile.write_text(''.join(json.dumps(m,sort_keys=True)+'\n' for m in logs))
    print(json.dumps({'lean_commit':lean_pin,'mathlib_commit':lib_pin,'pinned_mathlib_clean':True,
                      'build_flags':FLAGS,'source_sha256':sources,'fresh_object_sha256':objects,
                      'source_and_separate_import_theorems':len(imported),'new_theorems':10,
                      'all_observed_axioms':sorted({v for ax in imported.values() for v in ax}),
                      'theorem_log_sha256':h(logfile),'input_reviewer_script_sha256':h(HERE/'verify_ordinary_safe_modules.py')},
                     sort_keys=True,indent=2))


if __name__=='__main__':main()
