def ln_bounds(x, n):
    x=Q(x); t=(x-1)/(x+1)
    s=2*sum(t**(2*j+1)/(2*j+1) for j in range(n))
    err=2*abs(t)**(2*n+1)/((2*n+1)*(1-t*t))
    return s-err,s+err

def rows():
    with gzip.open(BASE/'FIFTH_WITNESSES.txt.gz','rt') as f:
        it=iter(f.read().split());take=lambda: int(next(it)); n,L,K,cp,rp,nw,ne=[take() for _ in range(7)]
        mask=[]; coeff=[]
        for _ in range(nw):
            mask.append(next(it)); cnt=take(); d={}
            for j in range(cnt):
                k,s,t,c=take(),take(),take(),take();assert (k,s,t) not in d;d[k,s,t]=c
            coeff.append(d)
        edges=[tuple(take() for j in range(4)) for _ in range(ne)]
        assert next(it,None) is None
        return (n,L,K,cp,rp),mask,coeff,edges

def cells(a,num):
    points=sorted(((Q(-j*a)%1,j) for j in range(num+1)))
    assert len({f for f,j in points})==num+1
    cuts=[f for f,j in points]+[1]
    def seq(f):return ''.join(str(1+((f+(j+1)*a)//1-(f+j*a)//1)) for j in range(num))
    words=[seq((x+y)/2) for x,y in zip(cuts,cuts[1:])]; assert words==[seq(x) for x in cuts[:-1]]
    return [j for f,j in points],words,[y-x for x,y in zip(cuts,cuts[1:])]

def rational_tail(l):
    r=Q(56,125); J=80-l; v=Q(5,2)-l
    co=Q(1); absolute=Q(1)
    for j in range(1,J+1):co*=abs(v-j+1)*r/j;absolute+=co
    first=co*abs(v-J)*r/(J+1)
    tail=first/(1-r*(1+(abs(v)+1)/(J+2)))
    return Q(2619,1000)**l * Q(5,4)**ceil(v)*(2*absolute+tail)*tail

    for (i,j,m,R),w in zip(edges,ls):assert mask[i]==w[:-1] and mask[j]==w[1:] and m==int(w[0]);assert 2< Q(R,2**rp)<6
    logs=[2*u2+ln_bounds(Q(R,4*2**rp),24)[1] for _,_,_,R in edges]
    avg=max(sum(f*x for f,x in zip(weights,logs)) for weights in (lw,hw));critical=4*l3/3

    mag=(1-(pi*80/252)**2/2)**-2;assert mag<4
    grid_path=BASE/'FIFTH_INTEGER_GRID_ROWS.txt'

    bounds=[[int(v) for v in row.split()] for row in raw];assert len(bounds)==77 and all(len(row)==138 for row in bounds)
