"""Rebuild both root-authored ordinary-gate modules from source without parent objects."""
from hashlib import sha256
from pathlib import Path
from tempfile import TemporaryDirectory
import json
import os
import re
import subprocess

HERE=Path(__file__).resolve().parent
BASE=Path('/workspace/scratch/9067acf67ee8/toolchain')
LEAN=BASE/'lean-4.34.1-linux/bin/lean'
MATHLIB=BASE/'mathlib4'
ORIGINAL=HERE.parent/'sixhour-root-wave5'
NAMES=('PermanentSafeDebt','PermanentSafeOrdinary')
FLAGS=['-j1','-DwarningAsError=true','-DautoImplicit=false','--json']
EXTRA={'propext','Classical.choice','Quot.sound'}


def run(args,wd,env=None):
    out=subprocess.run(args,cwd=wd,env=env,capture_output=True,text=True)
    if out.returncode!=0 or out.stderr:raise RuntimeError(out.stdout+out.stderr)
    return out.stdout


def messages(raw,required):
    result={}
    for row in raw.splitlines():
        msg=json.loads(row)
        if msg['severity']!='information':raise ValueError(msg)
        s=' '.join(msg['data'].split())
        a=re.fullmatch(r"'([.\w]+)' depends on axioms: \[(.*?)\]",s)
        b=re.fullmatch(r"'([.\w]+)' does not depend on any axioms",s)
        if a:
            name=a[1];ax=sorted(a[2].split(', ')) if a[2] else []
        elif b:
            name=b[1];ax=[]
        else:raise ValueError(s)
        if set(ax)-EXTRA or name in result:raise ValueError(s)
        result[name]=ax
    if set(result)!=set(required):raise ValueError((set(result),required))
    return result


def h(path):return sha256(path.read_bytes()).hexdigest()


def main():
    pin=run([LEAN,'--githash'],HERE).strip()
    lib=run(['git','-C',MATHLIB,'rev-parse','HEAD'],HERE).strip()
    if pin!='5045d0056413266e57c625dcd7c365b10e377c52' or lib!='d13f23b723b8a846827a245b89c10fc7d3f11612':
        raise ValueError((pin,lib))
    if run(['git','-C',MATHLIB,'status','--porcelain','--untracked-files=no'],HERE).strip():
        raise ValueError('modified Mathlib checkout')
    report={};all_decls={};logs=[]
    with TemporaryDirectory(dir=HERE) as temporary:
        wd=Path(temporary)
        if list(wd.iterdir()):raise ValueError('working directory is not fresh')
        env=os.environ.copy()
        paths=[wd,MATHLIB/'.lake/build/lib/lean',*sorted((MATHLIB/'.lake/packages').glob('*/.lake/build/lib/lean'))]
        env['LEAN_PATH']=os.pathsep.join(map(str,paths))
        for name in NAMES:
            src=wd/(name+'.lean')
            src.write_bytes((ORIGINAL/src.name).read_bytes())
            data=src.read_text()
            if re.search(r'\b(?:sorry|admit|axiom|native_decide|unsafe)\b',data):
                raise ValueError('disallowed trust/source token')
            declarations={name+'.'+n for n in re.findall(r'^(?:private\s+)?(?:theorem|lemma)\s+(\w+)',data,re.M)}
            printed=set(re.findall(r'^#print axioms ([.\w]+)',data,re.M))
            if declarations!=printed:raise ValueError('undeclared coverage')
            out=wd/(name+'.olean')
            raw=run([LEAN,*FLAGS,'-R',wd,'-o',out,src],wd,env)
            if not out.is_file():raise ValueError('missing fresh object')
            ax=messages(raw,declarations);all_decls.update(ax)
            logs.extend([{'stage':'source_build','module':name,'theorem':key,'axioms':val} for key,val in sorted(ax.items())])
            report[name]={'source_sha256':h(src),'object_sha256':h(out),'theorems':len(ax)}
        probe=wd/'OwnImport.lean'
        probe.write_text('import PermanentSafeOrdinary\n'+'\n'.join('#print axioms '+x for x in sorted(all_decls))+'\n')
        fresh=messages(run([LEAN,*FLAGS,'-R',wd,probe],wd,env),all_decls)
        if fresh!=all_decls:raise ValueError('separate source import mismatch')
        logs.extend([{'stage':'fresh_separate_import','theorem':k,'axioms':v} for k,v in sorted(fresh.items())])
    logfile=HERE/'ORDINARY_SAFE_LEAN_AUDIT_LOG.jsonl'
    logfile.write_text(''.join(json.dumps(m,sort_keys=True)+'\n' for m in logs))
    print(json.dumps({'lean':pin,'mathlib':lib,'flags':FLAGS,'mathlib_clean':True,
                      'modules_rebuilt_in_order':report,'separate_fresh_import':True,
                      'axioms_observed':sorted(set(x for v in fresh.values() for x in v)),
                      'theorem_count':len(fresh),'audit_log':h(logfile)},indent=2,sort_keys=True))


if __name__=='__main__':main()
