"""Fresh independent build and import of the root finite rank interface module."""
from tempfile import TemporaryDirectory
from pathlib import Path
import json
import os
import re
from verify_ordinary_safe_modules import HERE,ORIGINAL,LEAN,MATHLIB,FLAGS,run,messages,h


def main():
    name='RankConvexityInterfaces'
    lp=run([LEAN,'--githash'],HERE).strip()
    mp=run(['git','-C',MATHLIB,'rev-parse','HEAD'],HERE).strip()
    if (lp,mp)!=('5045d0056413266e57c625dcd7c365b10e377c52','d13f23b723b8a846827a245b89c10fc7d3f11612'):
        raise ValueError('dependency version')
    if run(['git','-C',MATHLIB,'status','--porcelain','--untracked-files=no'],HERE).strip():
        raise ValueError('modified Mathlib')
    with TemporaryDirectory(dir=HERE) as d:
        work=Path(d);src=work/(name+'.lean');obj=work/(name+'.olean')
        if list(work.iterdir()):raise ValueError('not initially empty')
        src.write_bytes((ORIGINAL/src.name).read_bytes())
        s=src.read_text()
        if re.search(r'\b(?:sorry|admit|axiom|native_decide|unsafe)\b',s):raise ValueError('trust marker')
        names={name+'.'+x for x in re.findall(r'^(?:private\s+)?(?:theorem|lemma)\s+(\w+)',s,re.M)}
        if names!=set(re.findall(r'^#print axioms ([.\w]+)',s,re.M)):raise ValueError('coverage')
        env=os.environ.copy()
        paths=[work,MATHLIB/'.lake/build/lib/lean',*sorted((MATHLIB/'.lake/packages').glob('*/.lake/build/lib/lean'))]
        env['LEAN_PATH']=os.pathsep.join(map(str,paths))
        a=messages(run([LEAN,*FLAGS,'-R',work,'-o',obj,src],work,env),names)
        probe=work/'IndependentRankImport.lean'
        probe.write_text('import '+name+'\n'+'\n'.join('#print axioms '+x for x in sorted(names))+'\n')
        b=messages(run([LEAN,*FLAGS,'-R',work,probe],work,env),names)
        if a!=b:raise ValueError('fresh import mismatch')
        print(json.dumps({'source_sha256':h(src),'fresh_object_sha256':h(obj),'flags':FLAGS,'lean':lp,
                          'mathlib':mp,'mathlib_clean':True,'fresh_import_confirmed':True,
                          'every_public_or_private_theorem':a},sort_keys=True,indent=2))


if __name__=='__main__':main()
