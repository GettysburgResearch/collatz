def T(x):
    assert x > 0
    return (3*x + 1)//2 if x % 2 else x//2


def v2(x):
    assert x > 0
    return (x & -x).bit_length() - 1


def trace(x, word):
    states = [x]
    for a in word:
        assert x > 0 and x % 2
        z = 3*x + 1
        assert v2(z) == a
        x = z // (2**a)
        states.append(x)
    return states


def target_window(b, H):
    assert (486*b + 102) % 2**H == 0
    x = n = 16384*b + 3431
    for j in range(1, 14+H):
        x = T(x)
        assert x > n, (b, H, j)
    return x
